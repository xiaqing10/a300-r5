cmake_minimum_required(VERSION 3.8)
project(a300_obstacle_avoidance)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
add_executable(a300_safety_controller src/a300_safety_controller.cpp)
ament_target_dependencies(a300_safety_controller rclcpp sensor_msgs geometry_msgs std_msgs nav_msgs)
install(TARGETS a300_safety_controller DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
ament_package()
